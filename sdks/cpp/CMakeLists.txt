cmake_minimum_required(VERSION 3.18)
project(l2s1_cpp VERSION 0.2.3 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
# Offline builds can point at a preinstalled nlohmann single-header directory.
set(L2S1_JSON_INCLUDE_DIR "" CACHE PATH "Directory containing nlohmann/json.hpp")
if(L2S1_JSON_INCLUDE_DIR)
    add_library(nlohmann_json::nlohmann_json INTERFACE IMPORTED)
    set_target_properties(nlohmann_json::nlohmann_json PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${L2S1_JSON_INCLUDE_DIR}")
else()
    find_package(nlohmann_json 3.11 REQUIRED)
endif()
add_library(l2s1_cpp src/l2s1.cpp)
add_library(l2s1::cpp ALIAS l2s1_cpp)
set_target_properties(l2s1_cpp PROPERTIES EXPORT_NAME cpp)
target_compile_features(l2s1_cpp PUBLIC cxx_std_17)
target_include_directories(l2s1_cpp PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(l2s1_cpp PUBLIC $<BUILD_INTERFACE:nlohmann_json::nlohmann_json> $<INSTALL_INTERFACE:nlohmann_json::nlohmann_json> PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(l2s1_cpp PRIVATE /W4 /permissive-)
else()
    target_compile_options(l2s1_cpp PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(L2S1_CPP_BUILD_EXAMPLES "Build C++ examples" ON)
if(L2S1_CPP_BUILD_EXAMPLES)
    add_executable(l2s1_cpp_example examples/decide.cpp)
    target_link_libraries(l2s1_cpp_example PRIVATE l2s1::cpp)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(l2s1_cpp_fake_engine tests/fake_engine.cpp)
    target_link_libraries(l2s1_cpp_fake_engine PRIVATE nlohmann_json::nlohmann_json)
    target_compile_features(l2s1_cpp_fake_engine PRIVATE cxx_std_17)
    add_executable(l2s1_cpp_tests tests/client.cpp)
    target_link_libraries(l2s1_cpp_tests PRIVATE l2s1::cpp)
    add_executable(l2s1_cpp_model_test tests/model.cpp)
    target_link_libraries(l2s1_cpp_model_test PRIVATE l2s1::cpp)
    add_test(NAME cpp_contract COMMAND l2s1_cpp_tests $<TARGET_FILE:l2s1_cpp_fake_engine>)
endif()
install(TARGETS l2s1_cpp EXPORT l2s1_cppTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT l2s1_cppTargets NAMESPACE l2s1:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/l2s1_cpp)
configure_package_config_file(cmake/l2s1_cppConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/l2s1_cppConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/l2s1_cpp)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/l2s1_cppConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/l2s1_cppConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/l2s1_cppConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/l2s1_cpp)
